import java.util.*;

public class SetBasics {

    public static void main(String[] args) {

        // ===== 1. Duplicates are simply ignored =====
        Set<String> accountNumbers = new HashSet<>();
        System.out.println("add AC1001 -> " + accountNumbers.add("AC1001"));
        System.out.println("add AC1002 -> " + accountNumbers.add("AC1002"));
        System.out.println("add AC1001 -> " + accountNumbers.add("AC1001") + "   <-- duplicate!");
        System.out.println("size       -> " + accountNumbers.size());

        // ===== 2. add()'s return value IS your duplicate check =====
        System.out.println("\n--- registering accounts ---");
        for (String candidate : List.of("AC1003", "AC1001", "AC1004", "AC1003")) {
            if (accountNumbers.add(candidate)) {
                System.out.println("  registered " + candidate);
            } else {
                System.out.println("  REJECTED " + candidate + " - already exists");
            }
        }

        // ===== 3. contains() is O(1), not a scan =====
        System.out.println("\ncontains AC1002? " + accountNumbers.contains("AC1002"));
        System.out.println("contains AC9999? " + accountNumbers.contains("AC9999"));

        // ===== 4. The three orderings, same input =====
        List<String> input = List.of("Rahul", "Priya", "Anita", "Priya", "Deepa", "Anita");
        System.out.println("\ninput            : " + input);
        System.out.println("HashSet          : " + new HashSet<>(input));
        System.out.println("LinkedHashSet    : " + new LinkedHashSet<>(input));
        System.out.println("TreeSet          : " + new TreeSet<>(input));

        // ===== 5. Dedupe a List, keeping the original order - one line =====
        List<String> deduped = new ArrayList<>(new LinkedHashSet<>(input));
        System.out.println("deduped, ordered : " + deduped);

        // ===== 6. Set operations - COPY FIRST, they mutate =====
        System.out.println("\n--- set operations ---");
        Set<String> javaDevs   = new HashSet<>(Set.of("Priya", "Rahul", "Anita"));
        Set<String> pythonDevs = new HashSet<>(Set.of("Anita", "Deepa", "Vikram"));

        Set<String> union = new TreeSet<>(javaDevs);
        union.addAll(pythonDevs);
        System.out.println("union        : " + union);

        Set<String> both = new TreeSet<>(javaDevs);
        both.retainAll(pythonDevs);
        System.out.println("intersection : " + both);

        Set<String> javaOnly = new TreeSet<>(javaDevs);
        javaOnly.removeAll(pythonDevs);
        System.out.println("java only    : " + javaOnly);

        System.out.println("javaDevs unchanged? " + new TreeSet<>(javaDevs));

        // ===== 7. TreeSet's navigation powers =====
        System.out.println("\n--- TreeSet range queries ---");
        TreeSet<Integer> scores = new TreeSet<>(List.of(45, 67, 72, 88, 91));
        System.out.println("all          : " + scores);
        System.out.println("first / last : " + scores.first() + " / " + scores.last());
        System.out.println("higher(72)   : " + scores.higher(72));
        System.out.println("floor(70)     : " + scores.floor(70));
        System.out.println("subSet(60,90): " + scores.subSet(60, 90));
        System.out.println("descending   : " + scores.descendingSet());
    }
}

import java.util.*;

public class BucketLab {

    // HashMap default capacity is 16. This is the same arithmetic the JDK uses.
    static final int CAPACITY = 16;

    public static void main(String[] args) {

        List<Account> accounts = List.of(
            new SavingsAccount     ("AC1001", "Priya Sharma",  50000),
            new CurrentAccount     ("AC1002", "Rahul Verma",   80000, 25000),
            new FixedDepositAccount("AC1003", "Anita Desai",  200000, 12),
            new SalaryAccount      ("AC1004", "Vikram Rao",     1200),
            new SavingsAccount     ("AC1005", "Deepa Iyer",    30000),
            new SavingsAccount     ("AC1009", "Arjun Mehta",   80000)
        );

        // ===== 1. Show the hash and the bucket each one lands in =====
        System.out.println("=== Which bucket does each account go to? ===");
        System.out.printf("%-10s %14s %12s %8s%n", "NUMBER", "hashCode()", "spread", "bucket");
        Map<Integer, List<String>> buckets = new TreeMap<>();

        for (Account a : accounts) {
            int hash   = a.hashCode();
            int spread = hash ^ (hash >>> 16);        // what HashMap actually does
            int bucket = spread & (CAPACITY - 1);      // cheap modulo for powers of 2
            System.out.printf("%-10s %14d %12d %8d%n",
                              a.getAccountNumber(), hash, spread, bucket);
            buckets.computeIfAbsent(bucket, k -> new ArrayList<>())
                   .add(a.getAccountNumber());
        }

        System.out.println("\n=== The bucket table ===");
        for (int i = 0; i < CAPACITY; i++) {
            List<String> contents = buckets.get(i);
            String shown = (contents == null) ? "null"
                         : String.join(" -> ", contents)
                           + (contents.size() > 1 ? "   <-- COLLISION, chained" : "");
            System.out.printf("  bucket %2d : %s%n", i, shown);
        }

        // ===== 2. The duplicate test - same number, different object =====
        System.out.println("\n=== Duplicate detection ===");
        Set<Account> set = new HashSet<>(accounts);
        System.out.println("set size after adding 6 accounts : " + set.size());

        Account impostor = new SavingsAccount("AC1001", "Someone Else", 999);
        System.out.println("\nA DIFFERENT object with the SAME number:");
        System.out.println("  equals existing?  " + impostor.equals(accounts.get(0)));
        System.out.println("  same hashCode?    "
                           + (impostor.hashCode() == accounts.get(0).hashCode()));
        System.out.println("  set.contains()?   " + set.contains(impostor));
        System.out.println("  set.add() ->      " + set.add(impostor) + "  (false = rejected)");
        System.out.println("  set size still    " + set.size());
    }
}

This is the explanation you were promised in Session 2 equals() still returns true — the objects are logically equal. But contains() computes the hash first, goes to that one bucket, finds nothing there, and returns false. equals() is never called. The lookup never gets far enough to ask the question.

That's why this bug is so dangerous: no exception, no warning, just a "unique" set holding duplicates and a contains() that lies. Restore hashCode() before moving on.
Optional but very illuminating — the terrible hashCode Temporarily change hashCode() to return 1; and run again. It's technically correct (equal objects still have equal hashes) so duplicate detection still works — but look at the bucket table: every account is chained into bucket 1. You've turned an O(1) structure into a linked list.
TASK 3▶ The Mutable Key TrapTheory slide 7
Goal Lose an object inside a HashSet — permanently — by changing a field after insertion. Short task, unforgettable lesson, and it explains why real-world keys are almost always immutable Strings.
📄MutableKey.javacreate this file
import java.util.*;

// A DELIBERATELY BAD design: a mutable field used in hashCode()
class Tag {
    private String name;                      // NOT final - that's the problem

    Tag(String name) { this.name = name; }

    void setName(String name) { this.name = name; }   // mutation!
    String getName()          { return name; }

    @Override public boolean equals(Object o) {
        if (this == o) return true;
        if (o == null || getClass() != o.getClass()) return false;
        return Objects.equals(name, ((Tag) o).name);
    }
    @Override public int hashCode() { return Objects.hash(name); }
    @Override public String toString() { return "Tag(" + name + ")";  }
}

public class MutableKey {

    public static void main(String[] args) {
        Set<Tag> tags = new HashSet<>();
        Tag java = new Tag("java");
        tags.add(java);

        System.out.println("--- before mutation ---");
        System.out.println("set          : " + tags);
        System.out.println("contains?    : " + tags.contains(java));
        System.out.println("hashCode     : " + java.hashCode());

        // Change a field that hashCode() depends on
        java.setName("kotlin");

        System.out.println("\n--- after mutation ---");
        System.out.println("set          : " + tags + "   <-- it's visibly IN there");
        System.out.println("contains?    : " + tags.contains(java) + "   <-- but not findable!");
        System.out.println("hashCode     : " + java.hashCode() + "   <-- moved bucket");
        System.out.println("size         : " + tags.size());
        System.out.println("remove()     : " + tags.remove(java) + "   <-- can't even delete it");
        System.out.println("size after   : " + tags.size() + "   <-- still there. Leaked forever.");

        // Only a full scan can still find it
        System.out.print("\nfound by iterating: ");
        for (Tag t : tags) if (t == java) System.out.print("YES - it really is in the set");
        System.out.println();

        // ===== THE FIX: an immutable key =====
        System.out.println("\n--- the fix: make the field final ---");
        System.out.println("If 'name' were final, setName() could not exist,");
        System.out.println("the hash could never change, and this bug is impossible.");
        System.out.println("That is exactly why String makes such a good key.");
    }
}

--- before mutation --- set : [Tag(java)] contains? : true hashCode : 3254818 --- after mutation --- set : [Tag(kotlin)] <-- it's visibly IN there contains? : false <-- but not findable! hashCode : -1125574399 <-- moved bucket size : 1 remove() : false <-- can't even delete it size after : 1 <-- still there. What actually happened The object was stored in the bucket for hash 3254818. Changing name changed its hash to -1125574399, so every lookup now searches a different bucket — one the object was never placed in. It's still physically in the set (you can print it and iterate to it) but it is unreachable by contains, remove, or anything else hash-based.

The rule: only use immutable fields in hashCode(). Make them final, and the compiler enforces it for you.

import java.util.*;

public class MapLab {

    public static void main(String[] args) {

        // ===== 1. Basic Map operations =====
        Map<String, Account> byNumber = new LinkedHashMap<>();

        List<Account> accounts = List.of(
            new SavingsAccount     ("AC1001", "Priya Sharma",  50000),
            new CurrentAccount     ("AC1002", "Rahul Verma",   80000, 25000),
            new FixedDepositAccount("AC1003", "Anita Desai",  200000, 12),
            new SalaryAccount      ("AC1004", "Vikram Rao",     1200),
            new SavingsAccount     ("AC1005", "Deepa Iyer",    30000)
        );
        for (Account a : accounts) byNumber.put(a.getAccountNumber(), a);

        System.out.println("=== Direct lookup ===");
        System.out.println("get(AC1003)      : " + byNumber.get("AC1003"));
        System.out.println("get(AC9999)      : " + byNumber.get("AC9999") + "   (null, not an exception)");
        System.out.println("containsKey      : " + byNumber.containsKey("AC1002"));
        System.out.println("size             : " + byNumber.size());

        // ===== 2. put() OVERWRITES - a common surprise =====
        System.out.println("\n=== put() overwrites ===");
        Account old = byNumber.put("AC1001", new SavingsAccount("AC1001", "REPLACED", 1));
        System.out.println("put returned the OLD value: " + old);
        System.out.println("map now holds             : " + byNumber.get("AC1001"));
        byNumber.put("AC1001", accounts.get(0));                 // put it back
        System.out.println("putIfAbsent (already there): "
                           + byNumber.putIfAbsent("AC1001", new SalaryAccount("AC1001", "X", 0)));
        System.out.println("still the original?        : " + byNumber.get("AC1001").getHolder());

        // ===== 3. The three iteration views =====
        System.out.println("\n=== keySet() ===");
        System.out.println("  " + byNumber.keySet());

        System.out.println("=== values() - total balance ===");
        double total = 0;
        for (Account a : byNumber.values()) total += a.getBalance();
        System.out.printf("  total across all accounts: %.2f%n", total);

        System.out.println("=== entrySet() - the efficient way when you need both ===");
        for (Map.Entry<String, Account> e : byNumber.entrySet()) {
            System.out.printf("  %-8s -> %-14s %12.2f%n",
                              e.getKey(), e.getValue().getHolder(), e.getValue().getBalance());
        }

        System.out.println("=== forEach ===");
        byNumber.forEach((k, v) -> System.out.println("  " + k + " = " + v.accountType()));

        // ===== 4. THE MEASUREMENT: List scan vs Map lookup =====
        System.out.println("\n=== 200,000 records: List vs Map ===");
        int n = 200_000;
        List<String> bigList = new ArrayList<>(n);
        Map<String, Integer> bigMap = new HashMap<>(n);
        for (int i = 0; i < n; i++) {
            bigList.add("AC" + i);
            bigMap.put("AC" + i, i);
        }
        String needle = "AC" + (n - 1);          // worst case: the very last one

        long t1 = System.nanoTime();
        boolean inList = bigList.contains(needle);
        long listTime = System.nanoTime() - t1;

        long t2 = System.nanoTime();
        boolean inMap = bigMap.containsKey(needle);
        long mapTime = System.nanoTime() - t2;

        System.out.printf("  List.contains()   : %,10d ns  (found=%b)%n", listTime, inList);
        System.out.printf("  Map.containsKey() : %,10d ns  (found=%b)%n", mapTime, inMap);
        System.out.printf("  Map is roughly %,dx faster%n",
                          mapTime == 0 ? listTime : listTime / Math.max(mapTime, 1));
    }
}

import java.util.*;

public class CountGroupLab {

    public static void main(String[] args) {

        // ================= COUNTING: word frequency =================
        String text = """
                the quick brown fox jumps over the lazy dog
                the dog barks and the fox runs
                the quick dog wins the race
                """;

        String[] words = text.toLowerCase().split("\\s+");

        // --- Version A: verbose, with a null check ---
        Map<String, Integer> countA = new HashMap<>();
        for (String w : words) {
            if (w.isBlank()) continue;
            Integer current = countA.get(w);
            countA.put(w, current == null ? 1 : current + 1);
        }

        // --- Version B: getOrDefault ---
        Map<String, Integer> countB = new HashMap<>();
        for (String w : words) {
            if (w.isBlank()) continue;
            countB.put(w, countB.getOrDefault(w, 0) + 1);
        }

        // --- Version C: merge - the idiomatic one-liner ---
        Map<String, Integer> countC = new HashMap<>();
        for (String w : words) {
            if (w.isBlank()) continue;
            countC.merge(w, 1, Integer::sum);
        }

        System.out.println("All three agree: " + (countA.equals(countB) && countB.equals(countC)));
        System.out.println("distinct words : " + countC.size());

        // --- Sort by count descending, then alphabetically ---
        System.out.println("\n=== Top 5 words ===");
        countC.entrySet().stream()
              .sorted(Map.Entry.<String, Integer>comparingByValue().reversed()
                          .thenComparing(Map.Entry.comparingByKey()))
              .limit(5)
              .forEach(e -> System.out.printf("  %-8s %d%n", e.getKey(), e.getValue()));

        // --- Words appearing only once ---
        System.out.println("\n=== Words appearing exactly once ===");
        List<String> unique = new ArrayList<>();
        countC.forEach((word, count) -> { if (count == 1) unique.add(word); });
        Collections.sort(unique);
        System.out.println("  " + unique);

        // --- A sorted view, for free ---
        System.out.println("\n=== Alphabetical (TreeMap) ===");
        System.out.println("  " + new TreeMap<>(countC));

        // ================= GROUPING: accounts by type =================
        System.out.println("\n\n=== Grouping accounts by type ===");
        List<Account> accounts = List.of(
            new SavingsAccount     ("AC1001", "Priya Sharma",  50000),
            new CurrentAccount     ("AC1002", "Rahul Verma",   80000, 25000),
            new FixedDepositAccount("AC1003", "Anita Desai",  200000, 12),
            new SalaryAccount      ("AC1004", "Vikram Rao",     1200),
            new SavingsAccount     ("AC1005", "Deepa Iyer",    30000),
            new SavingsAccount     ("AC1006", "Arjun Mehta",   80000),
            new CurrentAccount     ("AC1007", "Neha Kapoor",   45000, 10000)
        );

        // computeIfAbsent: create the inner list ONLY when first needed
        Map<String, List<Account>> byType = new TreeMap<>();
        for (Account a : accounts) {
            byType.computeIfAbsent(a.accountType(), k -> new ArrayList<>()).add(a);
        }

        for (Map.Entry<String, List<Account>> e : byType.entrySet()) {
            double subtotal = 0;
            for (Account a : e.getValue()) subtotal += a.getBalance();
            System.out.printf("%n%s  (%d account(s), subtotal %.2f)%n",
                              e.getKey(), e.getValue().size(), subtotal);
            for (Account a : e.getValue()) {
                System.out.printf("    %-8s %-14s %12.2f%n",
                                  a.getAccountNumber(), a.getHolder(), a.getBalance());
            }
        }

        // ================= Grouping by a computed key =================
        System.out.println("\n=== Grouped by balance band ===");
        Map<String, List<String>> byBand = new TreeMap<>();
        for (Account a : accounts) {
            String band = a.getBalance() >= 100000 ? "3. high (100k+)"
                        : a.getBalance() >=  40000 ? "2. medium (40k-100k)"
                        :                             "1. low (under 40k)";
            byBand.computeIfAbsent(band, k -> new ArrayList<>()).add(a.getHolder());
        }
        byBand.forEach((band, holders) -> System.out.println("  " + band + " : " + holders));

        // ================= A Map of Sets: unique holders per type =================
        System.out.println("\n=== Unique holders per account type (Map of Sets) ===");
        Map<String, Set<String>> holdersByType = new TreeMap<>();
        for (Account a : accounts) {
            holdersByType.computeIfAbsent(a.accountType(), k -> new TreeSet<>())
                         .add(a.getHolder());
        }
        holdersByType.forEach((type, holders) ->
            System.out.println("  " + type + " : " + holders));
    }
}

The two patterns to memorise Counting: map.merge(key, 1, Integer::sum) — "add 1, or start at 1".
Grouping: map.computeIfAbsent(key, k -> new ArrayList<>()).add(item) — "get the list for this key, creating it if needed, then add".

Note the return of computeIfAbsent is the value, which is why you can chain .add(item) straight onto it. Also note using TreeMap instead of HashMap gave sorted output for free — no sorting code at all.

        } catch (NumberFormatException e) {
            System.out.println("   bad number on line " + lineNo + ": " + e.getMessage());

import java.util.*;

public class Bank {

    private final String name;

    // MAP: account number -> Account. O(1) lookup.
    // LinkedHashMap so listing and saving keep a predictable order.
    private final Map<String, Account> accounts = new LinkedHashMap<>();

    // SET: every holder name, automatically deduplicated.
    // TreeSet so it is always sorted, for free.
    private final Set<String> holders = new TreeSet<>();

    public Bank(String name) {
        this.name = name;
    }

    // ===== The Set/Map uniqueness guarantee =====
    public void addAccount(Account account) throws DuplicateAccountException {
        Objects.requireNonNull(account, "account must not be null");
        if (accounts.containsKey(account.getAccountNumber())) {      // O(1) check
            throw new DuplicateAccountException(account.getAccountNumber());
        }
        accounts.put(account.getAccountNumber(), account);
        holders.add(account.getHolder());
    }

    // ===== O(1) lookup instead of scanning a list =====
    public Account find(String accountNumber) throws AccountNotFoundException {
        Account account = accounts.get(accountNumber);
        if (account == null) {
            throw new AccountNotFoundException(accountNumber);
        }
        return account;
    }

    public boolean removeAccount(String accountNumber) {
        Account removed = accounts.remove(accountNumber);
        if (removed == null) return false;
        // The holder may still own other accounts, so only drop the name if not
        boolean stillHasAccounts = accounts.values().stream()
                .anyMatch(a -> a.getHolder().equals(removed.getHolder()));
        if (!stillHasAccounts) holders.remove(removed.getHolder());
        return true;
    }

    // ===== Grouping, with computeIfAbsent =====
    public Map<String, List<Account>> groupByType() {
        Map<String, List<Account>> byType = new TreeMap<>();
        for (Account a : accounts.values()) {
            byType.computeIfAbsent(a.accountType(), k -> new ArrayList<>()).add(a);
        }
        return byType;
    }

    public double totalDeposits() {
        double total = 0;
        for (Account a : accounts.values()) total += a.getBalance();
        return total;
    }

    public void applyInterestToAll() {
        for (Account a : accounts.values()) a.applyInterest();     // POLYMORPHIC
    }

    // Sorted views, using Comparator chains from Session 3
    public List<Account> sortedByBalance() {
        List<Account> list = new ArrayList<>(accounts.values());
        list.sort(Comparator.comparingDouble(Account::getBalance).reversed());
        return list;
    }

    // DEFENSIVE COPIES (Session 3)
    public Collection<Account> getAccounts() { return List.copyOf(accounts.values()); }

    // NOTE: return a NEW TreeSet, not Set.copyOf(holders).
    // Set.copyOf() gives an immutable set with UNSPECIFIED iteration order, which
    // would silently throw away our sorting. A TreeSet copy stays sorted.
    public Set<String> getHolders()          { return new TreeSet<>(holders); }

    public int size()                        { return accounts.size(); }
    public String getName()                  { return name; }
}
Note on applyInterest() If your Session 2 Account.applyInterest() returns double, that's fine — just ignore the return value here, or change the loop to collect the totals. If you never wrote it, add public void applyInterest() { balance += balance * interestRate(); } to Account.
A subtle trap in getHolders() — worth pausing on Notice the comment in getHolders(). The obvious defensive copy would be Set.copyOf(holders) — and it compiles, runs, and silently breaks your sorting, because Set.copyOf() returns an immutable set whose iteration order is unspecified. Your carefully chosen TreeSet ordering is thrown away at the door.

List.copyOf() is safe (lists are ordered by definition), but for a sorted set you must copy into a new TreeSet. If option 6 prints your holders out of alphabetical order, this is why — a genuinely easy mistake to make, and a good example of why you should always check output rather than assume.

import java.io.IOException;
import java.nio.charset.StandardCharsets;
import java.nio.file.*;
import java.util.ArrayList;
import java.util.List;

public class BankStorage {

    private static final String HEADER = "type,number,holder,balance";
    private final Path file;

    public BankStorage(Path file) {
        this.file = file;
    }

    // ===================== SAVE: objects -> CSV =====================
    public void save(Bank bank) throws IOException {
        List<String> lines = new ArrayList<>();
        lines.add(HEADER);
        for (Account a : bank.getAccounts()) {
            lines.add(String.join(",",
                a.accountType(),
                a.getAccountNumber(),
                a.getHolder(),
                String.format("%.2f", a.getBalance())));
        }
        if (file.getParent() != null) Files.createDirectories(file.getParent());
        Files.write(file, lines, StandardCharsets.UTF_8);       // closes internally
    }

    // ===================== LOAD: CSV -> objects =====================
    public Bank load(String bankName) throws IOException {
        Bank bank = new Bank(bankName);

        if (!Files.exists(file)) {
            System.out.println("  (no save file yet - starting fresh)");
            return bank;                       // first run is NOT an error
        }

        List<String> lines = Files.readAllLines(file, StandardCharsets.UTF_8);
        int loaded = 0, skipped = 0;

        for (int i = 1; i < lines.size(); i++) {           // skip the header row
            String line = lines.get(i);
            if (line.isBlank()) continue;

            String[] p = line.split(",", -1);
            if (p.length != 4) {
                System.err.println("  line " + (i + 1) + " malformed, skipped: " + line);
                skipped++;
                continue;
            }
            try {
                double balance = Double.parseDouble(p[3].trim());

                // POLYMORPHISM: recreate the correct SUBCLASS from the type column
                Account account = switch (p[0].trim()) {
                    case "Savings"       -> new SavingsAccount(p[1].trim(), p[2].trim(), balance);
                    case "Current"       -> new CurrentAccount(p[1].trim(), p[2].trim(), balance, 25000);
                    case "Fixed Deposit" -> new FixedDepositAccount(p[1].trim(), p[2].trim(), balance, 12);
                    case "Salary"        -> new SalaryAccount(p[1].trim(), p[2].trim(), balance);
                    default -> throw new IllegalArgumentException("unknown type '" + p[0] + "'");
                };
                bank.addAccount(account);
                loaded++;

            } catch (NumberFormatException e) {
                System.err.println("  line " + (i + 1) + " bad balance, skipped");
                skipped++;
            } catch (IllegalArgumentException e) {
                System.err.println("  line " + (i + 1) + " " + e.getMessage() + ", skipped");
                skipped++;
            } catch (DuplicateAccountException e) {
                System.err.println("  line " + (i + 1) + " " + e.getMessage() + ", skipped");
                skipped++;
            }
        }
        System.out.println("  loaded " + loaded + " account(s), skipped " + skipped);
        return bank;
    }
}

▶ Common Issues & Fixes
SymptomCause & fix
My file is empty / truncatedThe writer was never closed, so the buffer was never flushed. Use try-with-resources.
HashSet contains duplicatesMissing or wrong hashCode(). It must use the same fields as equals().
contains() false for an object that's in thereEither no hashCode(), or you mutated a field used by it after insertion.
NullPointerException adding to a TreeSetTreeSet rejects null — it can't compare it. Use HashSet, or filter nulls out.
ClassCastException in a TreeSet/TreeMapYour class doesn't implement Comparable. Implement it, or pass a Comparator to the constructor.
map.get(key) returns null unexpectedlyThe key's equals/hashCode don't match, or there's whitespace. trim() keys parsed from files.
ConcurrentModificationException on a MapYou modified the map while iterating a view. Use entrySet().removeIf(...).
FileNotFoundException and you're sure it existsRelative paths resolve from the working directory, not the source folder. Print Path.of("x").toAbsolutePath() to see where it's actually looking.
NoSuchFileException from Files.writeThe parent directory doesn't exist. Call Files.createDirectories(path.getParent()) first.
AccessDeniedExceptionNo write permission there. Choose a folder you own.
Accented/Hindi/₹ characters are mangledCharset mismatch. Specify StandardCharsets.UTF_8 on both the writer and the reader.
ArrayIndexOutOfBoundsException when parsing CSVA short or blank row. Check parts.length before indexing, and use split(",", -1).
CSV row splits into too many fieldsA comma inside a data value (e.g. a name). Reject commas on input, or use a proper CSV library.
