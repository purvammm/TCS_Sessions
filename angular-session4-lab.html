  { path: '',           redirectTo: '/employees', pathMatch: 'full' },
  { path: 'employees',  component: EmployeeListComponent },
  { path: 'register',   component: RegistrationComponent },
  { path: 'reactive',   component: ReactiveRegistrationComponent },
  { path: 'api-users',  component: UserListComponent },

  { path: '**',         component: NotFoundComponent }    // MUST be last

@NgModule({
  imports: [RouterModule.forRoot(routes)],
  exports: [RouterModule]
})
export class AppRoutingModule { }

import { Component, OnInit } from '@angular/core';
import { Customer } from '../models/customer';

@Component({
  selector: 'app-customer-management',
  templateUrl: './customer-management.component.html',
  styleUrls: ['./customer-management.component.css']
})
export class CustomerManagementComponent implements OnInit {

  // the MASTER list — never filtered in place
  customers: Customer[] = [
    { id: 1, name: 'Anoop',  age: 40, mobile: '9876543210' },
    { id: 2, name: 'Ajay',   age: 32, mobile: '9123456780' },
    { id: 3, name: 'Rekha',  age: 28, mobile: '9988776655' },
    { id: 4, name: 'Suresh', age: 45, mobile: '9001122334' }
  ];

  // what the table renders — a VIEW of the master list
  filtered: Customer[] = [];

  searchTerm: string = '';
  message: string = '';

  // the insert form is bound to this object
  newCustomer: Customer = { id: 0, name: '', age: 0, mobile: '' };

  ngOnInit(): void {
    this.filtered = this.customers;
  }

  // ---------- SEARCH ----------
  onSearch(): void {
    const term = this.searchTerm.toLowerCase().trim();
    if (term === '') {
      this.filtered = this.customers;
      return;
    }
    this.filtered = this.customers.filter(c =>
      c.name.toLowerCase().includes(term) || c.mobile.includes(term)
    );
  }

  clearSearch(): void {
    this.searchTerm = '';
    this.onSearch();
  }

  // ---------- INSERT ----------
  insertCustomer(): void {
    if (!this.newCustomer.name.trim() || !this.newCustomer.mobile.trim()) {
      this.message = '⚠ Name and mobile are required.';
      return;
    }
    if (this.newCustomer.age < 1 || this.newCustomer.age > 120) {
      this.message = '⚠ Please enter a realistic age.';
      return;
    }

    // next id = highest existing + 1 (never reuse a deleted id)
    const nextId = this.customers.length === 0
      ? 1
      : Math.max(...this.customers.map(c => c.id)) + 1;

    // spread = push a COPY, so the form can't keep editing the new row
    this.customers.push({ ...this.newCustomer, id: nextId });

    this.onSearch();
    this.resetForm();
    this.message = '✓ Customer added with id ' + nextId + '.';
  }

  resetForm(): void {
    this.newCustomer = { id: 0, name: '', age: 0, mobile: '' };
  }

  // ---------- DELETE ----------
  deleteCustomer(c: Customer): void {
    const ok = confirm('Delete customer ' + c.id + ' — ' + c.name + '?');
    if (!ok) {
      this.message = 'Delete cancelled.';
      return;
    }

    // find by ID in the MASTER list, not by index in the filtered view
    const index = this.customers.findIndex(x => x.id === c.id);
    if (index !== -1) {
      this.customers.splice(index, 1);
    }

    this.onSearch();
    this.message = '✓ Customer ' + c.id + ' deleted.';
  }
}

Test the two subtle bugs this code avoids — genuinely worth 3 minutes

① Add to the component class

📄src/app/customer-management/customer-management.component.tsADD — inside the class, after deleteCustomer
  // ---------- UPDATE ----------

  // null when not editing; a COPY of the row when editing
  editCustomer: Customer | null = null;

  selectForEdit(c: Customer): void {
    // edit a COPY so Cancel genuinely discards the changes
    this.editCustomer = { ...c };
    this.message = 'Editing customer ' + c.id + ' — ID and Name are locked.';
  }

  saveEdit(): void {
    if (!this.editCustomer) { return; }

    if (this.editCustomer.age < 1 || this.editCustomer.age > 120) {
      this.message = '⚠ Please enter a realistic age.';
      return;
    }
    if (!this.editCustomer.mobile.trim()) {
      this.message = '⚠ Mobile cannot be empty.';
      return;
    }

    const index = this.customers.findIndex(x => x.id === this.editCustomer!.id);
    if (index !== -1) {
      this.customers[index] = { ...this.editCustomer };   // write the copy back
    }

    this.message = '✓ Customer ' + this.editCustomer.id + ' updated.';
    this.editCustomer = null;      // close the panel
    this.onSearch();               // refresh the view
  }

  cancelEdit(): void {
    this.editCustomer = null;
    this.message = 'Edit cancelled — nothing was changed.';
  }
What's that ! in this.editCustomer!.id? The non-null assertion operator. Inside the arrow function TypeScript can't prove editCustomer is still non-null, even though we checked it three lines up. The ! says "trust me, it isn't null here". Use it sparingly and only when you've genuinely already checked — it switches off a safety feature.

Prove why we edit a copy (the most instructive 60 seconds in this lab) Change selectForEdit from this.editCustomer = { ...c }; to this.editCustomer = c; and save. Now click Edit, change the age, and press Cancel. The table row has already changed. There was nothing left to cancel, because you were editing the live object the table renders. Put the spread back — and remember this pattern for every edit form you ever build.
Why [value] + readonly instead of [(ngModel)]? [value] is one-way: the value is displayed, with no path back into the object. readonly stops the user typing. Using [(ngModel)] on a field you call "read-only" is a contradiction — and if someone later removes the readonly attribute, edits would silently start flowing through. Make the binding itself express the intent.

① The service — the only file that knows the URL

📄src/app/customer.service.tsREPLACE the whole file
import { Injectable } from '@angular/core';
import { HttpClient } from '@angular/common/http';
import { Observable } from 'rxjs';
import { Customer } from './models/customer';

@Injectable({ providedIn: 'root' })
export class CustomerService {

  // A free fake REST API. Swap this ONE line for your own backend, e.g.
  //   private baseUrl = 'http://localhost:8080/api/customers';
  private baseUrl = 'https://jsonplaceholder.typicode.com/users';

  constructor(private http: HttpClient) { }

  /** READ — all customers */
  getAll(): Observable<Customer[]> {
    return this.http.get<Customer[]>(this.baseUrl);
  }

  /** READ — one customer */
  getById(id: number): Observable<Customer> {
    return this.http.get<Customer>(`${this.baseUrl}/${id}`);
  }

  /** CREATE — the SERVER assigns the id */
  add(customer: Customer): Observable<Customer> {
    return this.http.post<Customer>(this.baseUrl, customer);
  }

  /** UPDATE */
  update(customer: Customer): Observable<Customer> {
    return this.http.put<Customer>(`${this.baseUrl}/${customer.id}`, customer);
  }

  /** DELETE */
  remove(id: number): Observable<any> {
    return this.http.delete(`${this.baseUrl}/${id}`);
  }
}
About the test API's data shape JSONPlaceholder's /users returns objects with id, name, email, phone — so age will be undefined and mobile comes from a different field. That's fine and realistic: the API rarely matches your model exactly. The component below maps the response into Customer shape, which is precisely what you'd do in production.

② The component — now asynchronous

📄src/app/customer-management/customer-management.component.tsREPLACE the whole file
import { Component, OnInit } from '@angular/core';
import { Customer } from '../models/customer';
import { CustomerService } from '../customer.service';

@Component({
  selector: 'app-customer-management',
  templateUrl: './customer-management.component.html',
  styleUrls: ['./customer-management.component.css']
})
export class CustomerManagementComponent implements OnInit {

  customers: Customer[] = [];
  filtered: Customer[] = [];

  searchTerm: string = '';
  message: string = '';
  loading: boolean = false;

  newCustomer: Customer = { id: 0, name: '', age: 0, mobile: '' };
  editCustomer: Customer | null = null;

  // INJECT the service — no 'new', no URL in this file
  constructor(private customerService: CustomerService) { }

  ngOnInit(): void {
    this.loadAll();
  }

  // ---------- READ ----------
  loadAll(): void {
    this.loading = true;
    this.customerService.getAll().subscribe({
      next: (data: any[]) => {
        // map the API's shape into OUR model
        this.customers = data.map(u => ({
          id: u.id,
          name: u.name,
          age: u.age ?? 30,                        // the API has no age
          mobile: (u.phone || u.mobile || '')      // it calls it 'phone'
                    .toString().replace(/[^0-9]/g, '').slice(0, 10)
        }));
        this.onSearch();
        this.loading = false;
        this.message = '✓ Loaded ' + this.customers.length + ' customers from the API.';
      },
      error: (err) => {
        this.loading = false;
        this.message = '✖ Could not load customers.';
        console.error(err);
      }
    });
  }

  // ---------- SEARCH (still local — filtering in-memory is correct here) ----------
  onSearch(): void {
    const term = this.searchTerm.toLowerCase().trim();
    this.filtered = term === ''
      ? this.customers
      : this.customers.filter(c =>
          c.name.toLowerCase().includes(term) || c.mobile.includes(term));
  }

  clearSearch(): void {
    this.searchTerm = '';
    this.onSearch();
  }

  // ---------- CREATE ----------
  insertCustomer(): void {
    if (!this.newCustomer.name.trim() || !this.newCustomer.mobile.trim()) {
      this.message = '⚠ Name and mobile are required.';
      return;
    }

    this.customerService.add(this.newCustomer).subscribe({
      next: (created) => {
        // the SERVER decided the id — use what it returned
        this.customers.push({ ...this.newCustomer, id: created.id });
        this.onSearch();
        this.resetForm();
        this.message = '✓ Created via POST — server assigned id ' + created.id + '.';
      },
      error: (err) => {
        this.message = '✖ Create failed.';
        console.error(err);
      }
    });
  }

  resetForm(): void {
    this.newCustomer = { id: 0, name: '', age: 0, mobile: '' };
  }

  // ---------- UPDATE ----------
  selectForEdit(c: Customer): void {
    this.editCustomer = { ...c };
    this.message = 'Editing customer ' + c.id + ' — ID and Name are locked.';
  }

  saveEdit(): void {
    if (!this.editCustomer) { return; }
    const toSave = { ...this.editCustomer };

    this.customerService.update(toSave).subscribe({
      next: () => {
        const i = this.customers.findIndex(x => x.id === toSave.id);
        if (i !== -1) { this.customers[i] = toSave; }
        this.editCustomer = null;
        this.onSearch();
        this.message = '✓ Updated via PUT — customer ' + toSave.id + '.';
      },
      error: (err) => {
        this.message = '✖ Update failed.';
        console.error(err);
      }
    });
  }

  cancelEdit(): void {
    this.editCustomer = null;
    this.message = 'Edit cancelled — nothing was changed.';
  }

  // ---------- DELETE ----------
  deleteCustomer(c: Customer): void {
    if (!confirm('Delete customer ' + c.id + ' — ' + c.name + '?')) {
      this.message = 'Delete cancelled.';
      return;
    }

    this.customerService.remove(c.id).subscribe({
      next: () => {
        const i = this.customers.findIndex(x => x.id === c.id);
        if (i !== -1) { this.customers.splice(i, 1); }
        this.onSearch();
        this.message = '✓ Deleted via DELETE — customer ' + c.id + '.';
      },
      error: (err) => {
        this.message = '✖ Delete failed.';
        console.error(err);
      }
    });
  }
}

import { Component, OnInit } from '@angular/core';
import { HttpClient } from '@angular/common/http';

interface Post { id: number; title: string; body: string; }

@Component({
  selector: 'app-post-search',
  templateUrl: './post-search.component.html'
})
export class PostSearchComponent implements OnInit {

  allPosts: Post[] = [];
  results: Post[] = [];
  term: string = '';
  loading = false;

  constructor(private http: HttpClient) { }

  ngOnInit(): void {
    this.loading = true;
    this.http.get<Post[]>('https://jsonplaceholder.typicode.com/posts')
      .subscribe({
        next: (data) => { this.allPosts = data; this.results = data.slice(0, 10); this.loading = false; },
        error: (e) => { console.error(e); this.loading = false; }
      });
  }

  search(): void {
    const t = this.term.toLowerCase().trim();
    this.results = t === ''
      ? this.allPosts.slice(0, 10)
      : this.allPosts.filter(p => p.title.toLowerCase().includes(t)).slice(0, 20);
  }
}

The real-world version of this For a large dataset you'd query the server on every keystroke instead — but firing a request per character is wasteful. The professional solution uses RxJS operators: debounceTime(300) (wait until typing pauses), distinctUntilChanged() (ignore repeats) and switchMap() (cancel the previous in-flight request).
