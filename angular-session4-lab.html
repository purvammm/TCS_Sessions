Today's three golden rules 1. A standalone component imports what its own template uses. RouterOutlet, RouterLink, ReactiveFormsModule, CurrencyPipe — each in the imports: [ ] array of the component that uses it. This is the single biggest difference from older Angular tutorials.
2. Route order matters. Angular takes the first matching route, not the best one. The wildcard ** goes last, and static paths go above dynamic ones.
3. Never mutate a signal's value. list().push(x) changes nothing on screen. Always .set() or .update() with a new array. Rules 2 and 3 fail silently — rule 1 at least gives you an error.

Goal Turn one endless scrolling page into four separate screens with real URLs. By the end of this task, typing localhost:4200/about into the address bar will open the About screen — and the Back button will work.

LineWhy it's there
export const routes: RoutesA plain exported array. No class, no @NgModule, no decorator.
path: '' with pathMatch: 'full'pathMatch: 'full' means "only when the whole URL is empty". Leave it out and the empty path matches the start of every URL — an infinite redirect loop.
path: 'home' (no leading slash)Angular adds the slash. Writing '/home' gives you a route that never matches.
path: '**'The wildcard. Matches anything, so it must be last — Angular takes the first match, not the best.
provideRouter(routes, ...)The whole of the old AppRoutingModule, in one line. Called once, here.
withComponentInputBinding()Optional router feature. Turn it on now so Task 5 works; you can ignore it until then.

Try it yourself — six experiments, each proving one rule from the theory
  1. Prove the imports array is real. Delete RouterOutlet from the imports array (leave the tag in the template). The build fails with "'router-outlet' is not a known element" and the message even suggests importing it. Put it back. This is the error you will hit most often this session, so meet it on purpose now.
  2. Prove URLs are real. Type localhost:4200/about straight into the address bar and press Enter. The About screen opens directly. That is why routing matters — the URL is now a genuine address you can bookmark or send to someone.
  3. Prove routerLink is not href. Change the About link to <a href="/about">. Click it and watch the browser's reload spinner — the entire app restarts. Change it back to routerLink and the difference is obvious.
  4. Prove the outlet is required. Comment out <router-outlet />. Now the URL still changes when you click, and nothing appears at all — with no error anywhere. Angular has a component and nowhere to put it. Restore it.
  5. Prove route order matters. Move the { path: '**' } route to the top of the array. Now every single page shows the 404, because every URL matches the wildcard first. Move it back to the bottom. This is the single most common routing bug there is.
  6. Prove exact: true is needed. Remove [routerLinkActiveOptions]="{ exact: true }" from the Home link, then change that link's path from /home to /. It is highlighted on every page, because prefix matching means / matches everything. Put both back.

📄 src/app/pages/customers/customers.component.ts Replace the whole file 
import { Component, computed, inject, signal } from '@angular/core';
import { FormBuilder, ReactiveFormsModule, Validators } from '@angular/forms';
import { CurrencyPipe } from '@angular/common';
import { Customer } from '../../models/customer';

@Component({
  selector: 'app-customers',
  // The template uses [formGroup] and the currency pipe, so both come in here.
  imports: [ReactiveFormsModule, CurrencyPipe],
  templateUrl: './customers.component.html',
  styleUrl: './customers.component.css'
})
export class CustomersComponent {

  // ── the data: ONE source of truth ───────────────────────
  readonly allCustomers = signal<Customer[]>([
    { id: 1001, name: 'Anoop Kumar',  city: 'Ahmedabad', email: 'anoop@example.com',  balance: 15000 },
    { id: 1002, name: 'Ajay Sharma',  city: 'Pune',      email: 'ajay@example.com',   balance: 0 },
    { id: 1003, name: 'Rekha Nair',   city: 'Kochi',     email: 'rekha@example.com',  balance: 42500 },
    { id: 1004, name: 'Suresh Patel', city: 'Surat',     email: 'suresh@example.com', balance: 8200 }
  ]);

  // ── screen state ────────────────────────────────────────
  readonly searchTerm = signal('');
  readonly message    = signal('');

  // ── edit mode: null = adding, a number = editing that id ─
  readonly editingId = signal<number | null>(null);

  // ── DERIVED state — this is what the table renders ───────
  // Recomputed automatically whenever allCustomers or searchTerm changes.
  // There is no onSearch() method, and no second array to keep in step.
  readonly customers = computed(() => {
    const t = this.searchTerm().trim().toLowerCase();
    if (t === '') { return this.allCustomers(); }
    return this.allCustomers().filter(c =>
      c.name.toLowerCase().includes(t) ||
      c.city.toLowerCase().includes(t) ||
      c.email.toLowerCase().includes(t)
    );
  });

  readonly totalBalance = computed(() =>
    this.customers().reduce((sum, c) => sum + c.balance, 0)
  );

  // ── the form (reactive, from Session 3) ─────────────────
  private fb = inject(FormBuilder);

  readonly custForm = this.fb.nonNullable.group({
    name:    ['', [Validators.required, Validators.minLength(3)]],
    city:    ['', [Validators.required]],
    email:   ['', [Validators.required, Validators.email]],
    balance: [0,  [Validators.required, Validators.min(0)]]
  });

  // Typed forms give us .controls with no possible undefined,
  // so the template needs no ?. at all.
  get name()    { return this.custForm.controls.name; }
  get city()    { return this.custForm.controls.city; }
  get email()   { return this.custForm.controls.email; }
  get balance() { return this.custForm.controls.balance; }

  // ── R: SEARCH ───────────────────────────────────────────
  clearSearch(): void {
    this.searchTerm.set('');     // the computed() does the rest
  }

  // ── C: INSERT (U comes in Task 4) ───────────────────────
  onSubmit(): void {
    if (this.custForm.invalid) {
      this.custForm.markAllAsTouched();   // reveal every error at once
      return;
    }

    const newCustomer: Customer = {
      id: this.nextId(),
      ...this.custForm.getRawValue()      // fully typed: name, city, email, balance
    };

    // IMMUTABLE update: a NEW array, not push(). Signals compare by reference.
    this.allCustomers.update(list => [...list, newCustomer]);

    this.message.set('✅ Added ' + newCustomer.name + ' (id ' + newCustomer.id + ').');
    this.resetForm();
  }

  // ── D: DELETE ───────────────────────────────────────────
  onDelete(customer: Customer): void {
    const ok = confirm('Delete ' + customer.name + '? This cannot be undone.');
    if (!ok) { return; }

    this.allCustomers.update(list => list.filter(c => c.id !== customer.id));

    if (this.editingId() === customer.id) {
      this.resetForm();   // don't stay in edit mode on something that's gone
    }

    this.message.set('🗑️ Deleted ' + customer.name + '.');
  }

  // ── helpers ─────────────────────────────────────────────
  private nextId(): number {
    const list = this.allCustomers();
    if (list.length === 0) { return 1001; }
    // highest existing id + 1 — NOT length + 1, which duplicates after a delete
    return Math.max(...list.map(c => c.id)) + 1;
  }

  resetForm(): void {
    this.custForm.reset({ name: '', city: '', email: '', balance: 0 });
    this.editingId.set(null);
  }
}
DecisionWhy it's written that way
customers = computed(...)The big one. The displayed list is derived, not stored. It cannot drift out of step with the data, so there is no onSearch() to call after a change and no chance of filtering the wrong array.
update(list => [...list, x])Signals detect change by reference. list.push(x) keeps the same reference, so nothing would re-render. Always produce a new array.
editingId = signal<number | null>One value carries the whole insert-versus-update distinction. Task 4 uses it; Session 1's Stretch C is where you met | null.
fb.nonNullable.groupTyped forms: getRawValue() returns real strings and numbers, and controls.name is never undefined — so no ?. in the template.
markAllAsTouched()Without it, clicking Save on an untouched empty form does nothing and shows nothing — the user has no idea why.
Math.max(...ids) + 1Highest id plus one. length + 1 produces duplicate ids the moment anything is deleted — and duplicates break @for's track.
confirm(...) before deleteNever destroy data on a single click. The browser dialog is ugly but honest — real projects use a styled modal with identical logic.

import { Injectable, signal } from '@angular/core';
import { Customer } from '../models/customer';

@Injectable({ providedIn: 'root' })
export class CustomerService {

  // PRIVATE and writable — only this class may change the list.
  private readonly _customers = signal<Customer[]>([
    { id: 1001, name: 'Anoop Kumar',  city: 'Ahmedabad', email: 'anoop@example.com',  balance: 15000 },
    { id: 1002, name: 'Ajay Sharma',  city: 'Pune',      email: 'ajay@example.com',   balance: 0 },
    { id: 1003, name: 'Rekha Nair',   city: 'Kochi',     email: 'rekha@example.com',  balance: 42500 },
    { id: 1004, name: 'Suresh Patel', city: 'Surat',     email: 'suresh@example.com', balance: 8200 }
  ]);

  // PUBLIC and read-only — components can read it, but cannot call .set().
  readonly customers = this._customers.asReadonly();

  // ── R ───────────────────────────────────────────────────
  getById(id: number): Customer | undefined {
    return this._customers().find(c => c.id === id);
  }

  // ── C ───────────────────────────────────────────────────
  /** Assigns the next id and returns the customer that was actually stored. */
  add(customer: Customer): Customer {
    const created: Customer = { ...customer, id: this.nextId() };
    this._customers.update(list => [...list, created]);
    return created;
  }

  // ── U ───────────────────────────────────────────────────
  /** Returns false if there was no such customer — so the caller can be honest. */
  update(customer: Customer): boolean {
    if (!this.getById(customer.id)) { return false; }
    this._customers.update(list =>
      list.map(c => c.id === customer.id ? customer : c));
    return true;
  }

  // ── D ───────────────────────────────────────────────────
  delete(id: number): boolean {
    if (!this.getById(id)) { return false; }
    this._customers.update(list => list.filter(c => c.id !== id));
    return true;
  }

  // ── helpers ─────────────────────────────────────────────
  private nextId(): number {
    const list = this._customers();
    if (list.length === 0) { return 1001; }
    return Math.max(...list.map(c => c.id)) + 1;
  }
}
Look at those method names — they were chosen deliberately getById, add, update, delete. That is the shape of a repository, and it is why Task 7 can swap the array for HTTP calls without renaming a single thing. Notice too that update and delete return a boolean — "did I find it?" — so the caller can react honestly instead of assuming.
asReadonly() is the whole point of this step Before: any component could call customers.set([]) and wipe the list. After: the compiler refuses. Every change now goes through a method the service owns — which is where you would later add validation, logging, an audit trail, or an HTTP call. That last one is Task 7, and it is only easy because of this step.

import { inject, Injectable } from '@angular/core';
import { HttpClient } from '@angular/common/http';
import { Observable } from 'rxjs';
import { map } from 'rxjs/operators';
import { Customer } from '../models/customer';

@Injectable({ providedIn: 'root' })
export class CustomerService {

  private http = inject(HttpClient);

  // ONE line to change when you switch to your own backend.
  private baseUrl = 'https://jsonplaceholder.typicode.com/users';

  // ── R: all ──────────────────────────────────────────────
  getAll(): Observable<Customer[]> {
    return this.http.get<any[]>(this.baseUrl).pipe(
      map(users => users.map(u => this.toCustomer(u)))
    );
  }

  // ── R: one ──────────────────────────────────────────────
  getById(id: number): Observable<Customer> {
    return this.http.get<any>(`${this.baseUrl}/${id}`).pipe(
      map(u => this.toCustomer(u))
    );
  }

  // ── C ───────────────────────────────────────────────────
  add(customer: Customer): Observable<Customer> {
    return this.http.post<Customer>(this.baseUrl, customer);
  }

  // ── U ───────────────────────────────────────────────────
  update(customer: Customer): Observable<Customer> {
    return this.http.put<Customer>(`${this.baseUrl}/${customer.id}`, customer);
  }

  // ── D ───────────────────────────────────────────────────
  delete(id: number): Observable<void> {
    return this.http.delete<void>(`${this.baseUrl}/${id}`);
  }

  /** The API's shape is not our shape. All the conversion lives here. */
  private toCustomer(u: any): Customer {
    return {
      id: u.id,
      name: u.name,
      city: u.address?.city ?? 'Unknown',
      email: u.email,
      balance: u.id * 1000
    };
  }
}

import { toObservable, toSignal } from '@angular/core/rxjs-interop';
import { of } from 'rxjs';
import { catchError, switchMap } from 'rxjs/operators';

readonly customer = toSignal(
  toObservable(this.id).pipe(
    switchMap(id => this.customerService.getById(id).pipe(
      catchError(() => of(undefined))       // a 404 becomes "not found"
    ))
  )
);
